# HTTP/RTSP response parsing known answers. `raw` is the wire bytes with
# \r and \n escaped; the parser is also fed one byte at a time. Optional:
# `head` (response to HEAD), `headers` (k=v | k=v), `body`, `rest` (bytes
# left after the message), `error` (ErrorKind).

name = RTSP with Content-Length and pipelined rest
raw = RTSP/1.0 200 OK\r\nContent-Length: 5\r\nCSeq: 1\r\n\r\nhelloRTSP/1.0
version = RTSP/1.0
status = 200
reason = OK
headers = Content-Length=5 | CSeq=1
body = hello
rest = RTSP/1.0

reason = OK
